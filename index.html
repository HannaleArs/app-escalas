  <script>

    const audioContext = new (window.AudioContext || window.webkitAudioContext)();

    const gainNode = audioContext.createGain();

    gainNode.connect(audioContext.destination);


    const waveTypeSelect = document.getElementById('waveType');

    const volumeSlider = document.getElementById('volume');

    const keyboard = document.getElementById('keyboard');


    const notes = [

      { note: 'C3', freq: 130.81, type: 'white' },

      { note: 'C#3', freq: 138.59, type: 'black' },

      { note: 'E3', freq: 164.81, type: 'white' },

      { note: 'F3', freq: 174.61, type: 'white' },

      { note: 'F#3', freq: 185.00, type: 'black' },

      { note: 'G3', freq: 196.00, type: 'white' },

      { note: 'G#3', freq: 207.65, type: 'black' },

      { note: 'A3', freq: 220.00, type: 'white' },

      { note: 'A#3', freq: 233.08, type: 'black' },

      { note: 'B3', freq: 246.94, type: 'white' },


      { note: 'C4', freq: 261.63, type: 'white' },

      { note: 'C#4', freq: 277.18, type: 'black' },

      { note: 'D4', freq: 293.66, type: 'white' },

      { note: 'D#4', freq: 311.13, type: 'black' },

      { note: 'E4', freq: 329.63, type: 'white' },

      { note: 'F4', freq: 349.23, type: 'white' },

      { note: 'F#4', freq: 369.99, type: 'black' },

      { note: 'G4', freq: 392.00, type: 'white' },

      { note: 'G#4', freq: 415.30, type: 'black' },

      { note: 'A4', freq: 440.00, type: 'white' },

      { note: 'A#4', freq: 466.16, type: 'black' },

      { note: 'B4', freq: 493.88, type: 'white' }

    ];


    let lastWhiteKey = null;


    notes.forEach((n) => {

      if (n.type === 'white') {

        const whiteKey = document.createElement('div');

        whiteKey.className = 'key white';

        whiteKey.dataset.freq = n.freq;

        whiteKey.textContent = n.note;

        keyboard.appendChild(whiteKey);

        lastWhiteKey = whiteKey;

      } else if (n.type === 'black' && lastWhiteKey) {

        const blackKey = document.createElement('div');

        blackKey.className = 'key black';

        blackKey.dataset.freq = n.freq;

        blackKey.textContent = n.note;

        lastWhiteKey.appendChild(blackKey);

      }

    });


    function playSound(freq) {

      if (audioContext.state === 'suspended') {

        audioContext.resume();

      }


      const oscillator = audioContext.createOscillator();

      const envelope = audioContext.createGain();


      oscillator.type = waveTypeSelect.value;

      oscillator.frequency.setValueAtTime(freq, audioContext.currentTime);


      const now = audioContext.currentTime;

      const currentVol = parseFloat(volumeSlider.value);


      envelope.gain.setValueAtTime(0, now);

      envelope.gain.linearRampToValueAtTime(currentVol, now + 0.02);

      envelope.gain.exponentialRampToValueAtTime(0.0001, now + 0.6);


      oscillator.connect(envelope);

      envelope.connect(gainNode);


      oscillator.start(now);

      oscillator.stop(now + 0.6);

    }


    function playScale(scaleNotes) {

      scaleNotes.forEach((note, index) => {

        if(!note) return;

        setTimeout(() => {

          let key = keyboard.querySelector(`[data-freq="${note.freq}"]`);

          if (key) {

            key.classList.add('active');

            setTimeout(() => key.classList.remove('active'), 400);

          }

          playSound(note.freq);

        }, index * 500);

      });

    }


    keyboard.addEventListener('mousedown', (e) => {

      if (e.target.classList.contains('key')) {

        const freq = parseFloat(e.target.dataset.freq);

        playSound(freq);

      }

    });


    volumeSlider.addEventListener('input', (e) => {

      gainNode.gain.value = e.target.value;

    });


    document.getElementById('play-major').addEventListener('click', () => {

      const majorScale = [

        notes.find(n => n.note === 'C3'),

        notes.find(n => n.note === 'D3'),

        notes.find(n => n.note === 'E3'),

        notes.find(n => n.note === 'G3'),

        notes.find(n => n.note === 'A3')

      ];

      playScale(majorScale);

    });


    document.getElementById('play-minor').addEventListener('click', () => {

      const minorScale = [

        notes.find(n => n.note === 'C3'),

        notes.find(n => n.note === 'D#3'),

        notes.find(n => n.note === 'F3'),

        notes.find(n => n.note === 'G3'),

        notes.find(n => n.note === 'A#3')

      ];

      playScale(minorScale);

    });


    let score = 0;

    let currentChallenge = null;


    document.getElementById('start-challenge').addEventListener('click', () => {

      const randomScaleType = Math.random() > 0.5 ? 'major' : 'minor';

      const randomTonicIndex = Math.floor(Math.random() * 12); 

      const tonic = notes[randomTonicIndex];


      if (randomScaleType === 'major') {

        currentChallenge = {

          type: 'major',

          scale: [

            tonic,

            notes[randomTonicIndex + 2],

            notes[randomTonicIndex + 4],

            notes[randomTonicIndex + 7],

            notes[randomTonicIndex + 9]

          ]

        };

      } else {

        currentChallenge = {

          type: 'minor',

          scale: [

            tonic,

            notes[randomTonicIndex + 3],

            notes[randomTonicIndex + 5],

            notes[randomTonicIndex + 7],

            notes[randomTonicIndex + 10]

          ]

        };

      }


      playScale(currentChallenge.scale);


      const msg = document.getElementById('challenge-message');

      msg.style.color = '#0056b3'; // Azul inicial

      msg.textContent = `Escucha la escala e intenta adivinar si es Mayor o Menor.`;

    });


    document.getElementById('guess-major').addEventListener('click', () => {

      if (!currentChallenge) return;

      const msg = document.getElementById('challenge-message');


      if (currentChallenge.type === 'major') {

        score++;

        msg.style.color = '#28a745'; // VERDE para acierto

        msg.textContent = `¡Correcto! Era una Pentatónica Mayor iniciada en ${currentChallenge.scale[0].note}. Puntos: ${score}`;

      } else {

        msg.style.color = '#dc3545'; // ROJO para fallo

        msg.textContent = `Incorrecto. Era una Pentatónica Menor iniciada en ${currentChallenge.scale[0].note}. Puntos: ${score}`;

      }

      currentChallenge = null;

    });


    document.getElementById('guess-minor').addEventListener('click', () => {

      if (!currentChallenge) return;

      const msg = document.getElementById('challenge-message');


      if (currentChallenge.type === 'minor') {

        score++;

        msg.style.color = '#28a745'; // VERDE para acierto

        msg.textContent = `¡Correcto! Era una Pentatónica Menor iniciada en ${currentChallenge.scale[0].note}. Puntos: ${score}`;

      } else {

        msg.style.color = '#dc3545'; // ROJO para fallo

        msg.textContent = `Incorrecto. Era una Pentatónica Mayor iniciada en ${currentChallenge.scale[0].note}. Puntos: ${score}`;

      }

      currentChallenge = null;

    });
